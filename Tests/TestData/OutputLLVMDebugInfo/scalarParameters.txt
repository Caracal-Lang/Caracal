define i32 @describe(i32 %0, float %1, i1 %2, ptr %3) !dbg !4 {

  store i32 %0, ptr %count, align 4, !dbg !13

  %call = call i32 @puts(ptr %load2), !dbg !19
  %load3 = load i32, ptr %count, align 4, !dbg !20
  ret i32 %load3, !dbg !20

  ret i32 0, !dbg !21
}
