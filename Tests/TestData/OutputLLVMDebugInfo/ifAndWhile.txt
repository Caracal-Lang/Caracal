define i32 @main() !dbg !4 {
entry:
  %count = alloca i32, align 4, !dbg !9
    #dbg_declare(ptr %count, !10, !DIExpression(), !9)
  %total = alloca i32, align 4, !dbg !11
    #dbg_declare(ptr %total, !12, !DIExpression(), !11)
  store i32 0, ptr %count, align 4, !dbg !9
  store i32 0, ptr %total, align 4, !dbg !11
  br label %while.condition, !dbg !13

while.condition:                                  ; preds = %while.body, %entry
  %load = load i32, ptr %count, align 4, !dbg !13
  %less_than = icmp slt i32 %load, 3, !dbg !13
  br i1 %less_than, label %while.body, label %while.continuation, !dbg !13

while.body:                                       ; preds = %while.condition
  %load1 = load i32, ptr %total, align 4, !dbg !14
  %load2 = load i32, ptr %count, align 4, !dbg !14
  %add = add i32 %load1, %load2, !dbg !14
  store i32 %add, ptr %total, align 4, !dbg !14
  %load3 = load i32, ptr %count, align 4, !dbg !15
  %add4 = add i32 %load3, 1, !dbg !15
  store i32 %add4, ptr %count, align 4, !dbg !15
  br label %while.condition, !dbg !13

while.continuation:                               ; preds = %while.condition
  %load5 = load i32, ptr %total, align 4, !dbg !16
  %equal = icmp eq i32 %load5, 3, !dbg !16
  br i1 %equal, label %if.true, label %if.continuation, !dbg !16

if.true:                                          ; preds = %while.continuation
  ret i32 0, !dbg !17

if.continuation:                                  ; preds = %while.continuation
  ret i32 1, !dbg !18
}
