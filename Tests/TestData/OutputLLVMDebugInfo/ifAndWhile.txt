define i32 @main() !dbg !4 {
entry:
  br label %while.condition, !dbg !7

while.condition:                                  ; preds = %while.body, %entry
  %phi = phi i32 [ 0, %entry ], [ %add2, %while.body ], !dbg !7
  %phi1 = phi i32 [ 0, %entry ], [ %add, %while.body ], !dbg !7
  %less_than = icmp slt i32 %phi, 3, !dbg !7
  br i1 %less_than, label %while.body, label %while.continuation, !dbg !7

while.body:                                       ; preds = %while.condition
  %add = add i32 %phi1, %phi, !dbg !8
  %add2 = add i32 %phi, 1, !dbg !9
  br label %while.condition, !dbg !7

while.continuation:                               ; preds = %while.condition
  %equal = icmp eq i32 %phi1, 3, !dbg !10
  br i1 %equal, label %if.true, label %if.continuation, !dbg !10

if.true:                                          ; preds = %while.continuation
  ret i32 0, !dbg !11

if.continuation:                                  ; preds = %while.continuation
  ret i32 1, !dbg !12
}
