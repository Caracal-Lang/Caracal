  %count = alloca i32, align 4, !dbg !9

  store i32 0, ptr %count, align 4, !dbg !9
  store i64 0, ptr %total, align 8, !dbg !11
  store float 5.000000e-01, ptr %ratio, align 4, !dbg !14
  store i8 7, ptr %small, align 1, !dbg !17
  store i1 false, ptr %done, align 1, !dbg !20
  br label %while.condition, !dbg !23

while.condition:                                  ; preds = %while.body, %entry
  %load = load i1, ptr %done, align 1, !dbg !23
  %0 = xor i1 %load, true, !dbg !23
  br i1 %0, label %while.body, label %while.continuation, !dbg !23

while.body:                                       ; preds = %while.condition
  %load1 = load i32, ptr %count, align 4, !dbg !24
  %add = add i32 %load1, 1, !dbg !24
  store i32 %add, ptr %count, align 4, !dbg !24
  %load2 = load i64, ptr %total, align 8, !dbg !25
  %add3 = add i64 %load2, 10, !dbg !25
  store i64 %add3, ptr %total, align 8, !dbg !25
  %load4 = load float, ptr %ratio, align 4, !dbg !26
  %multiply = fmul float %load4, 2.000000e+00, !dbg !26
  store float %multiply, ptr %ratio, align 4, !dbg !26
  %load5 = load i32, ptr %count, align 4, !dbg !27
  %equal = icmp eq i32 %load5, 3, !dbg !27
  store i1 %equal, ptr %done, align 1, !dbg !27
  br label %while.condition, !dbg !23

while.continuation:                               ; preds = %while.condition
  %load6 = load i32, ptr %count, align 4, !dbg !28
  %subtract = sub i32 %load6, 3, !dbg !28
  ret i32 %subtract, !dbg !28
}
