%string = type { { ptr, i32, i32 } }

@0 = private unnamed_addr constant [6 x i8] c"hello\00", align 1
@greeting = constant %string { { ptr, i32, i32 } { ptr @0, i32 5, i32 5 } }

declare i32 @strcmp(ptr %0, ptr %1)

declare ptr @calloc(i64 %0, i64 %1)

declare ptr @realloc(ptr %0, i64 %1)

declare ptr @memmove(ptr %0, ptr %1, i64 %2)

define i32 @lengthOf(%string %0) {
entry:
  %text = alloca %string, align 8
  store %string %0, ptr %text, align 8
  %call = call i32 @string.length(ptr %text)
  ret i32 %call
}

define i32 @main() {
entry:
  %call = call i32 @string.length(ptr @greeting)
  %load = load %string, ptr @greeting, align 8
  %call1 = call i32 @lengthOf(%string %load)
  %add = add i32 %call, %call1
  %subtract = sub i32 %add, 10
  ret i32 %subtract
}

define i32 @string.length(ptr nonnull %0) {
entry:
  %_bytes = getelementptr inbounds nuw %string, ptr %0, i32 0, i32 0
  %length = getelementptr inbounds nuw { ptr, i32, i32 }, ptr %_bytes, i32 0, i32 1
  %load = load i32, ptr %length, align 4
  ret i32 %load
}

; uselistorder directives
uselistorder ptr @string.length, { 1, 0 }
