define i32 @func() {
entry:
  br label %while.condition

while.condition:                                  ; preds = %if.continuation, %entry
  %phi = phi i32 [ 0, %entry ], [ %add, %if.continuation ]
  br i1 true, label %while.body, label %while.continuation

while.body:                                       ; preds = %while.condition
  %add = add i32 %phi, 1
  %greater_or_equal = icmp sge i32 %add, 3
  br i1 %greater_or_equal, label %if.true, label %if.continuation

if.true:                                          ; preds = %while.body
  br label %while.continuation

if.continuation:                                  ; preds = %while.body
  br label %while.condition

while.continuation:                               ; preds = %if.true, %while.condition
  %phi1 = phi i32 [ %phi, %while.condition ], [ %add, %if.true ]
  ret i32 %phi1

; uselistorder directives
  uselistorder i32 %add, { 0, 2, 1 }
}
