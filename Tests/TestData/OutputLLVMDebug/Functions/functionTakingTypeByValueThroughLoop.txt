; ModuleID = 'CaracalModule'
source_filename = "CaracalModule"
target triple = "x86_64-unknown-none"

%Number = type { i32 }

declare i32 @strcmp(ptr %0, ptr %1)

declare ptr @calloc(i64 %0, i64 %1)

declare ptr @realloc(ptr %0, i64 %1)

declare ptr @memmove(ptr %0, ptr %1, i64 %2)

define i32 @readLoop(%Number %0, i32 %1) {
entry:
  %n = alloca %Number, align 8
  store %Number %0, ptr %n, align 4
  br label %while.condition

while.condition:                                  ; preds = %while.body, %entry
  %phi = phi i32 [ %1, %entry ], [ %subtract, %while.body ]
  %greater_than = icmp sgt i32 %phi, 0
  br i1 %greater_than, label %while.body, label %while.continuation

while.body:                                       ; preds = %while.condition
  %subtract = sub i32 %phi, 1
  br label %while.condition

while.continuation:                               ; preds = %while.condition
  %value = getelementptr inbounds nuw %Number, ptr %n, i32 0, i32 0
  %load = load i32, ptr %value, align 4
  ret i32 %load
}

define i32 @func() {
entry:
  %constructed = alloca %Number, align 8
  call void @Number.new(ptr %constructed, i32 6)
  %load = load %Number, ptr %constructed, align 4
  %call = call i32 @readLoop(%Number %load, i32 2)
  ret i32 %call
}

define void @Number.new(ptr nonnull %0, i32 %1) {
entry:
  %value = getelementptr inbounds nuw %Number, ptr %0, i32 0, i32 0
  store i32 %1, ptr %value, align 4
  ret void
}
