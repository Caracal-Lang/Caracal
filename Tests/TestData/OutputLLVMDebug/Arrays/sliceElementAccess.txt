define i32 @sum({ ptr, i32 } %0) {
entry:
  %temp = alloca { ptr, i32 }, align 8
  %temp3 = alloca { ptr, i32 }, align 8
  br label %while.condition

while.condition:                                  ; preds = %while.body, %entry
  %phi = phi i32 [ 0, %entry ], [ %add6, %while.body ]
  %phi1 = phi i32 [ 0, %entry ], [ %add, %while.body ]
  %phi2 = phi { ptr, i32 } [ %0, %entry ], [ %phi2, %while.body ]
  store { ptr, i32 } %phi2, ptr %temp, align 8
  %length = getelementptr inbounds nuw { ptr, i32 }, ptr %temp, i32 0, i32 1
  %load = load i32, ptr %length, align 4
  %less_than = icmp slt i32 %phi, %load
  br i1 %less_than, label %while.body, label %while.continuation

while.body:                                       ; preds = %while.condition
  store { ptr, i32 } %phi2, ptr %temp3, align 8
  %data = getelementptr inbounds nuw { ptr, i32 }, ptr %temp3, i32 0, i32 0
  %load4 = load ptr, ptr %data, align 8
  %element = getelementptr i32, ptr %load4, i32 %phi
  %load5 = load i32, ptr %element, align 4
  %add = add i32 %phi1, %load5
  %add6 = add i32 %phi, 1
  br label %while.condition

while.continuation:                               ; preds = %while.condition
  ret i32 %phi1

; uselistorder directives
  uselistorder { ptr, i32 } %phi2, { 1, 2, 0 }
}
