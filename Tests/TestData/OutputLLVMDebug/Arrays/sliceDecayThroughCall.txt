define i32 @total({ ptr, i32 } %0) {
entry:
  %temp = alloca { ptr, i32 }, align 8
  store { ptr, i32 } %0, ptr %temp, align 8
  %length = getelementptr inbounds nuw { ptr, i32 }, ptr %temp, i32 0, i32 1
  %load = load i32, ptr %length, align 4
  ret i32 %load
}

define i32 @use() {
entry:
  %numbers = alloca [3 x i32], align 4
  %element = getelementptr [3 x i32], ptr %numbers, i32 0, i32 0
  store i32 1, ptr %element, align 4
  %element1 = getelementptr [3 x i32], ptr %numbers, i32 0, i32 1
  store i32 2, ptr %element1, align 4
  %element2 = getelementptr [3 x i32], ptr %numbers, i32 0, i32 2
  store i32 3, ptr %element2, align 4
  %0 = insertvalue { ptr, i32 } poison, ptr %numbers, 0
  %slice = insertvalue { ptr, i32 } %0, i32 3, 1
  %call = call i32 @total({ ptr, i32 } %slice)
  ret i32 %call
}
