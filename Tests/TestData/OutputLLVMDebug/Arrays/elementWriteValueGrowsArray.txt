define i32 @grow(ptr nonnull %0) {
entry:
  br label %while.condition

while.condition:                                  ; preds = %while.body, %entry
  %phi = phi i32 [ 0, %entry ], [ %add, %while.body ]
  %less_than = icmp slt i32 %phi, 64
  br i1 %less_than, label %while.body, label %while.continuation

while.body:                                       ; preds = %while.condition
  %data = getelementptr inbounds nuw { ptr, i32, i32 }, ptr %0, i32 0, i32 0
  %length = getelementptr inbounds nuw { ptr, i32, i32 }, ptr %0, i32 0, i32 1
  %capacity = getelementptr inbounds nuw { ptr, i32, i32 }, ptr %0, i32 0, i32 2
  %load = load ptr, ptr %data, align 8
  %load1 = load i32, ptr %length, align 4
  %load2 = load i32, ptr %capacity, align 4
  %is_full = icmp eq i32 %load1, %load2
  %doubled = mul i32 %load2, 2
  %new_capacity = select i1 %is_full, i32 %doubled, i32 %load2
  %widened = sext i32 %new_capacity to i64
  %bytes = mul i64 %widened, 4
  %grown = call ptr @realloc(ptr %load, i64 %bytes)
  store ptr %grown, ptr %data, align 8
  store i32 %new_capacity, ptr %capacity, align 4
  %element = getelementptr i32, ptr %grown, i32 %load1
  store i32 %phi, ptr %element, align 4
  %new_length = add i32 %load1, 1
  store i32 %new_length, ptr %length, align 4
  %add = add i32 %phi, 1
  br label %while.condition

while.continuation:                               ; preds = %while.condition
  ret i32 7
}
