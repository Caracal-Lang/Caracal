declare i32 @strcmp(ptr %0, ptr %1)

declare ptr @calloc(i64 %0, i64 %1)

declare ptr @realloc(ptr %0, i64 %1)

declare ptr @memmove(ptr %0, ptr %1, i64 %2)

define void @func() {
entry:
  %x = alloca i32, align 4
  store i32 1, ptr %x, align 4
  %load = load i32, ptr %x, align 4
  %equal = icmp eq i32 %load, 2
  br i1 %equal, label %if.true, label %if.continuation

if.true:                                          ; preds = %entry
  ret void

if.continuation:                                  ; preds = %entry
  ret void
}
